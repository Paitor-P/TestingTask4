package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest1 {

    @Test(timeout = 1000)
    public void testPutAndGet() {
        LruCache<Integer, String> lruCache = new LruCache<>(3);
        lruCache.put(1, "A");
        lruCache.put(2, "B");
        lruCache.put(3, "C");
        assertEquals("A", lruCache.get(1));
        assertEquals("B", lruCache.get(2));
        assertEquals("C", lruCache.get(3));
    }

    @Test(timeout = 1000)
    public void testCapacityLimit() {
        LruCache<Integer, String> lruCache = new LruCache<>(2);
        lruCache.put(1, "A");
        lruCache.put(2, "B");
        lruCache.put(3, "C");
        assertEquals("B", lruCache.get(1));
        assertEquals("C", lruCache.get(2));
        assertNull(lruCache.get(3));
    }

    @Test(timeout = 1000)
    public void testEmptyCache() {
        LruCache<Integer, String> lruCache = new LruCache<>(3);
        assertNull(lruCache.get(1));
        assertNull(lruCache.get(2));
        assertNull(lruCache.get(3));
    }
}
