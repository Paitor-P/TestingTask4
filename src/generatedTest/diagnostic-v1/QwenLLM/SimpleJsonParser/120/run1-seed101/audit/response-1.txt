package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest1 {

    @Test(timeout = 1000)
    public void testParseEmptyString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(null, parser.parse(""));
    }

    @Test(timeout = 1000)
    public void testParseNullString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertNull(parser.parse(null));
    }

    @Test(timeout = 1000)
    public void testParseSingleQuotedString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals("hello", parser.parse("\"hello\""));
    }

    @Test(timeout = 1000)
    public void testParseDoubleQuotedString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals("world", parser.parse("\"world\""));
    }

    @Test(timeout = 1000)
    public void testParseInteger() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(42, parser.parse("42"));
    }

    @Test(timeout = 1000)
    public void testParseDouble() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(123.45, parser.parse("123.45"));
    }

    @Test(timeout = 1000)
    public void testParseArray() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(List.of(1, 2, 3), parser.parse("[1, 2, 3]"));
    }

    @Test(timeout = 1000)
    public void testParseNestedArray() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(List.of(List.of(1, 2), List.of(3, 4)), parser.parse("[[1, 2], [3, 4]]"));
    }

    @Test(timeout = 1000)
    public void testParseStringWithEscapes() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals("hello\\nworld", parser.parse("\"hello\\nworld\""));
    }
}
