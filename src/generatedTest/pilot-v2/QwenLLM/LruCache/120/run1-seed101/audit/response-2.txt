package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest2 {

    @Test(timeout = 1000)
    public void testBasicPutAndGet() {
        LruCache<Integer, String> lruCache = new LruCache<>(3);
        lruCache.put(1, "A");
        lruCache.put(2, "B");
        lruCache.put(3, "C");
        assertEquals("A", lruCache.get(1));
        assertEquals("B", lruCache.get(2));
        assertEquals("C", lruCache.get(3));
    }

    @Test(timeout = 1000)
    public void testEviction() {
        LruCache<Integer, String> lruCache = new LruCache<>(2);
        lruCache.put(1, "A");
        lruCache.put(2, "B");
        lruCache.put(3, "C");
        assertNull(lruCache.get(1)); // A is evicted
        assertEquals("B", lruCache.get(2));
        assertEquals("C", lruCache.get(3));
    }

    @Test(timeout = 1000)
    public void testOverwrite() {
        LruCache<Integer, String> lruCache = new LruCache<>(2);
        lruCache.put(1, "A");
        lruCache.put(2, "B");
        lruCache.put(1, "A1"); // B is evicted, A is overwritten
        assertEquals("A1", lruCache.get(1));
        assertNull(lruCache.get(2)); // B is not accessible
        assertEquals("C", lruCache.get(3)); // C is not affected
    }
}
