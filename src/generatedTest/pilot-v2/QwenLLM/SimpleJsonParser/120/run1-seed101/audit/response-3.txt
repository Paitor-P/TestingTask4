package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest3 {

    @Test(timeout = 1000)
    public void testParseEmptyString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertNull(parser.parse(""));
    }

    @Test(timeout = 1000)
    public void testParseNullString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertThrows(IllegalArgumentException.class, () -> parser.parse(null));
    }

    @Test(timeout = 1000)
    public void testParseSingleNumber() {
        SimpleJsonParser parser = new SimpleJsonParser();
        assertEquals(42.0, parser.parse("42"), 0.0001);
    }

    @Test(timeout = 1000)
    public void testParseArray() {
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = parser.parse("[1, 2, 3]");
        assertEquals(3, result.size());
        assertEquals(1, result.get(0));
        assertEquals(2, result.get(1));
        assertEquals(3, result.get(2));
    }

    @Test(timeout = 1000)
    public void testParseString() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String result = parser.parse("\"Hello, World!\"");
        assertEquals("Hello, World!", result);
    }

    @Test(timeout = 1000)
    public void testParseNestedArray() {
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = parser.parse("[[1, 2], [3, 4]]");
        assertEquals(2, result.size());
        List<Object> inner1 = (List<Object>) result.get(0);
        assertEquals(2, inner1.size());
        assertEquals(1, inner1.get(0));
        assertEquals(2, inner1.get(1));
        List<Object> inner2 = (List<Object>) result.get(1);
        assertEquals(2, inner2.size());
        assertEquals(3, inner2.get(0));
        assertEquals(4, inner2.get(1));
    }
}
