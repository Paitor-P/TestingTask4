package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest12 {

    @Test(timeout = 1000)
    public void testCalculateFinalPrice() {
        PricingRequest request1 = new PricingRequest(
                100.0,
                10,
                PricingEngine.CustomerType.REGULAR,
                true,
                "SAVE10",
                true,
                150
        );
        double expected1 = 100.0 * 10 * 0.85 * 0.95 + 7.99;
        double actual1 = PricingEngine.calculateFinalPrice(request1);
        assertEquals(expected1, actual1, 0.001);

        PricingRequest request2 = new PricingRequest(
                200.0,
                5,
                PricingEngine.CustomerType.VIP,
                false,
                "FLAT5",
                false,
                50
        );
        double expected2 = 200.0 * 5 * 0.88 * 0.95 + 7.99 - 5.0;
        double actual2 = PricingEngine.calculateFinalPrice(request2);
        assertEquals(expected2, actual2, 0.001);

        PricingRequest request3 = new PricingRequest(
                50.0,
                20,
                PricingEngine.CustomerType.REGULAR,
                true,
                "FREESHIP",
                false,
                100
        );
        double expected3 = 50.0 * 20 * 0.85 + 7.99;
        double actual3 = PricingEngine.calculateFinalPrice(request3);
        assertEquals(expected3, actual3, 0.001);

        PricingRequest request4 = new PricingRequest(
                100.0,
                20,
                PricingEngine.CustomerType.VIP,
                true,
                "SAVE10",
                true,
                200
        );
        double expected4 = 100.0 * 20 * 0.85 * 0.95 + 7.99;
        double actual4 = PricingEngine.calculateFinalPrice(request4);
        assertEquals(expected4, actual4, 0.001);
    }
}
