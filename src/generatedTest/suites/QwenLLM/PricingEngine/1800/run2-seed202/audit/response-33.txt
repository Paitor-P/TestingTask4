package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest33 {

    @Test(timeout=1000)
    public void testCalculateFinalPrice() {
        PricingRequest request = new PricingRequest(10.0, 5, PricingEngine.CustomerType.REGULAR, false, "SAVE10", false, 200);
        double result = new PricingEngine().calculateFinalPrice(request);
        assertEquals(48.2, result, 0.01); // Test case 1: Regular customer, SAVE10 coupon, 5 items, no seasonal sale

        request = new PricingRequest(10.0, 10, PricingEngine.CustomerType.VIP, false, "FLAT5", false, 0);
        result = new PricingEngine().calculateFinalPrice(request);
        assertEquals(86.0, result, 0.01); // Test case 2: VIP customer, FLAT5 coupon, 10 items, no seasonal sale, no loyalty points

        request = new PricingRequest(10.0, 15, PricingEngine.CustomerType.REGULAR, true, null, true, 0);
        result = new PricingEngine().calculateFinalPrice(request);
        assertEquals(136.89, result, 0.01); // Test case 3: Regular customer, seasonal sale, first order, no loyalty points

        request = new PricingRequest(20.0, 1, PricingEngine.CustomerType.VIP, true, "FREESHIP", false, 100);
        result = new PricingEngine().calculateFinalPrice(request);
        assertEquals(20.0, result, 0.01); // Test case 4: VIP customer, seasonal sale, first order, FREESHIP coupon, 1 item, 100 loyalty points

        request = new PricingRequest(15.0, 20, PricingEngine.CustomerType.VIP, false, "SAVE10", true, 500);
        result = new PricingEngine().calculateFinalPrice(request);
        assertEquals(281.0, result, 0.01); // Test case 5: VIP customer, SAVE10 coupon, 20 items, no seasonal sale, 500 loyalty points
    }
}
