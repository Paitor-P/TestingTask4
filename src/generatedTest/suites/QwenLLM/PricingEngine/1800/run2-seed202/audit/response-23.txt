package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.Arrays;

public class QwenTest23 {

    private PricingEngine pricingEngine = new PricingEngine();

    @Test(timeout = 1000)
    public void testCalculateFinalPrice() {
        // Test case 1: Basic case with no discounts or promotions
        PricingRequest request1 = new PricingRequest(
                100.0,
                10,
                PricingEngine.CustomerType.REGULAR,
                false,
                null,
                false,
                0
        );
        assertEquals(90.0, pricingEngine.calculateFinalPrice(request1), 0.001);

        // Test case 2: Regular customer with quantity >= 10
        PricingRequest request2 = new PricingRequest(
                100.0,
                10,
                PricingEngine.CustomerType.REGULAR,
                false,
                null,
                false,
                0
        );
        assertEquals(90.0, pricingEngine.calculateFinalPrice(request2), 0.001);

        // Test case 3: VIP customer with quantity >= 10 and seasonal sale
        PricingRequest request3 = new PricingRequest(
                100.0,
                10,
                PricingEngine.CustomerType.VIP,
                true,
                null,
                false,
                0
        );
        assertEquals(81.60, pricingEngine.calculateFinalPrice(request3), 0.001);

        // Test case 4: First order with loyalty points
        PricingRequest request4 = new PricingRequest(
                100.0,
                1,
                PricingEngine.CustomerType.REGULAR,
                false,
                null,
                true,
                100
        );
        assertEquals(105.50, pricingEngine.calculateFinalPrice(request4), 0.001);

        // Test case 5: Flat discount with coupon code
        PricingRequest request5 = new PricingRequest(
                50.0,
                1,
                PricingEngine.CustomerType.REGULAR,
                false,
                "FLAT5",
                false,
                0
        );
        assertEquals(45.0, pricingEngine.calculateFinalPrice(request5), 0.001);

        // Test case 6: No shipping with coupon code
        PricingRequest request6 = new PricingRequest(
                50.0,
                1,
                PricingEngine.CustomerType.REGULAR,
                false,
                "FREESHIP",
                false,
                0
        );
        assertEquals(50.0, pricingEngine.calculateFinalPrice(request6), 0.001);

        // Test case 7: Loyalty tier discount
        PricingRequest request7 = new PricingRequest(
                100.0,
                1,
                PricingEngine.CustomerType.REGULAR,
                false,
                null,
                false,
                100
        );
        assertEquals(95.0, pricingEngine.calculateFinalPrice(request7), 0.001);

        // Test case 8: VIP discount with quantity >= 20 and SAVE10 coupon
        PricingRequest request8 = new PricingRequest(
                100.0,
                20,
                PricingEngine.CustomerType.VIP,
                false,
                "SAVE10",
                false,
                0
        );
        assertEquals(84.60, pricingEngine.calculateFinalPrice(request8), 0.001);

        // Test case 9: Mixed discounts with coupon code
        PricingRequest request9 = new PricingRequest(
                100.0,
                10,
                PricingEngine.CustomerType.VIP,
                true,
                "SAVE10",
                true,
                500
        );
        assertEquals(80.88, pricingEngine.calculateFinalPrice(request9), 0.001);

        // Test case 10: Negative base price should throw IllegalArgumentException
        try {
            pricingEngine.calculateFinalPrice(new PricingRequest(
                    -100.0,
                    10,
                    PricingEngine.CustomerType.REGULAR,
                    false,
                    null,
                    false,
                    0
            ));
            fail("Expected IllegalArgumentException for negative base price");
        } catch (IllegalArgumentException e) {
            // Expected
        }
    }
}
