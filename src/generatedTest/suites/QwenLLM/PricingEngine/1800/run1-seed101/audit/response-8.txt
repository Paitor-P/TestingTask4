        PricingRequest request1 = new PricingRequest(100.0, 1, PricingEngine.CustomerType.REGULAR, false, "", false, 0);
        assertEquals(100.0 + 7.99, pricingEngine.calculateFinalPrice(request1), 0.01);

        // Test case 2: VIP customer, no discount, no coupon, no first order, no loyalty points
        PricingRequest request2 = new PricingRequest(100.0, 1, PricingEngine.CustomerType.VIP, false, "", false, 0);
        assertEquals(100.0 * 1.02 - 7.99, pricingEngine.calculateFinalPrice(request2), 0.01);

        // Test case 3: Seasonal sale, VIP customer, no discount, no coupon, no first order, no loyalty points
        PricingRequest request3 = new PricingRequest(100.0, 1, PricingEngine.CustomerType.VIP, true, "", false, 0);
        assertEquals(100.0 * 1.07 - 7.99, pricingEngine.calculateFinalPrice(request3), 0.01);

        // Test case 4: First order, regular customer, no discount, no coupon, no loyalty points
        PricingRequest request4 = new PricingRequest(100.0, 1, PricingEngine.CustomerType.REGULAR, false, "", true, 0);
