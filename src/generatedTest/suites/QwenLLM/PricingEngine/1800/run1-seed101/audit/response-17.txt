package com.viktor.lab4;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest17 {

    @Test(timeout = 1000)
    public void testCalculateFinalPrice() {
        PricingRequest request1 = new PricingRequest(
                100.0, 5, PricingEngine.CustomerType.REGULAR, false, "SAVE10", true, 200
        );
        double expected1 = 100.0 * 5 * 0.88 - 5.0 + 7.99;
        double result1 = PricingEngine.calculateFinalPrice(request1);
        assertEquals(expected1, result1, 0.001);

        PricingRequest request2 = new PricingRequest(
                150.0, 10, PricingEngine.CustomerType.VIP, true, "FLAT5", false, 50
        );
        double expected2 = 150.0 * 10 * 0.86 - 5.0 + 7.99;
        double result2 = PricingEngine.calculateFinalPrice(request2);
        assertEquals(expected2, result2, 0.001);

        PricingRequest request3 = new PricingRequest(
                50.0, 1, PricingEngine.CustomerType.REGULAR, false, null, false, 0
        );
        double expected3 = 50.0 * 1 * 0.95 + 7.99;
        double result3 = PricingEngine.calculateFinalPrice(request3);
        assertEquals(expected3, result3, 0.001);

        PricingRequest request4 = new PricingRequest(
                100.0, 20, PricingEngine.CustomerType.VIP, false, "SAVE10", true, 150
        );
        double expected4 = 100.0 * 20 * 0.88 - 5.0 + 7.99;
        double result4 = PricingEngine.calculateFinalPrice(request4);
        assertEquals(expected4, result4, 0.001);

        PricingRequest request5 = new PricingRequest(
                10.0, 10, PricingEngine.CustomerType.REGULAR, true, "FREESHIP", false, 0
        );
        double expected5 = 10.0 * 10 * 0.95 - 5.0;
        double result5 = PricingEngine.calculateFinalPrice(request5);
        assertEquals(expected5, result5, 0.001);
    }
}
