package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest30 {

    @Test(timeout = 1000)
    public void testInvalidInput() {
        SimpleJsonParser parser = new SimpleJsonParser("]");
        try {
            parser.parse("]");
            fail("Expected IllegalArgumentException for invalid input");
        } catch (IllegalArgumentException e) {
            assertEquals("Unexpected end of input", e.getMessage());
        }

        parser = new SimpleJsonParser("true");
        try {
            parser.parse("true");
            fail("Expected IllegalArgumentException for invalid input");
        } catch (IllegalArgumentException e) {
            assertEquals("Unsupported token 't'", e.getMessage());
        }

        parser = new SimpleJsonParser("a");
        try {
            parser.parse("a");
            fail("Expected IllegalArgumentException for invalid input");
        } catch (IllegalArgumentException e) {
            assertEquals("Unsupported token 'a'", e.getMessage());
        }
    }
}
