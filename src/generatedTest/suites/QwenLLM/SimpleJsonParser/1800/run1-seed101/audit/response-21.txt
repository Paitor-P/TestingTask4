package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.List;

public class QwenTest21 {

    @Test(timeout = 1000)
    public void testParseValidJsonArray() {
        String input = "[1, 2, 3, 4, 5]";
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = (List<Object>) parser.parse(input);
        assertNotNull(result);
        assertEquals(5, result.size());
        assertEquals(1, result.get(0));
        assertEquals(2, result.get(1));
        assertEquals(3, result.get(2));
        assertEquals(4, result.get(3));
        assertEquals(5, result.get(4));
    }

    @Test(timeout = 1000)
    public void testParseEmptyArray() {
        String input = "[]";
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = (List<Object>) parser.parse(input);
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }

    @Test(timeout = 1000)
    public void testParseSingleElementArray() {
        String input = "[1]";
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = (List<Object>) parser.parse(input);
        assertNotNull(result);
        assertEquals(1, result.size());
        assertEquals(1, result.get(0));
    }

    @Test(timeout = 1000)
    public void testParseMultipleEmptyStringElementsArray() {
        String input = "[\"\", \"\", \"\"]";
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = (List<Object>) parser.parse(input);
        assertNotNull(result);
        assertEquals(3, result.size());
        assertEquals("", result.get(0));
        assertEquals("", result.get(1));
        assertEquals("", result.get(2));
    }

    @Test(timeout = 1000)
    public void testParseNestedArray() {
        String input = "[[1, 2], [3, 4]]";
        SimpleJsonParser parser = new SimpleJsonParser();
        List<Object> result = (List<Object>) parser.parse(input);
        assertNotNull(result);
        assertEquals(2, result.size());

        List<Object> subArray1 = (List<Object>) result.get(0);
        assertEquals(2, subArray1.size());
        assertEquals(1, subArray1.get(0));
        assertEquals(2, subArray1.get(1));

        List<Object> subArray2 = (List<Object>) result.get(1);
        assertEquals(2, subArray2.size());
        assertEquals(3, subArray2.get(0));
        assertEquals(4, subArray2.get(1));
    }
}
