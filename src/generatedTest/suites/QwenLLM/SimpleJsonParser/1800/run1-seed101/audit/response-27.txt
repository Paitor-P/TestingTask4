import org.junit.Test;

    @Test(timeout = 1000)
    public void testValidJson() {
        String input = "{\"key\":\"value\", \"array\":[1, 2, 3], \"nested\": {\"deep\":\"nested\"}}";
        SimpleJsonParser parser = new SimpleJsonParser();
        Object parsed = parser.parse(input);

        assertTrue(parsed instanceof java.util.Map);
        Map<String, Object> map = (Map<String, Object>) parsed;
        assertEquals("value", map.get("key"));
        assertTrue(map.get("array") instanceof List);
        List<?> array = (List<?>) map.get("array");
        assertEquals(3, array.size());
        assertEquals(1, array.get(0));
        assertEquals(2, array.get(1));
        assertEquals(3, array.get(2));
        assertTrue(map.get("nested") instanceof Map);
        Map<String, Object> nestedMap = (Map<String, Object>) map.get("nested");
        assertEquals("nested", nestedMap.get("deep"));
    }
