package com.viktor.lab4;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest19 {

    @Test(timeout = 1000)
    public void testParseEmptyString() {
        SimpleJsonParser parser = new SimpleJsonParser("");
        assertNull(parser.parse(""));
    }

    @Test(timeout = 1000)
    public void testParseSingleQuotedString() {
        SimpleJsonParser parser = new SimpleJsonParser("\"hello\"");
        assertEquals("hello", parser.parse("\"hello\""));
    }

    @Test(timeout = 1000)
    public void testParseDoubleQuotedString() {
        SimpleJsonParser parser = new SimpleJsonParser("\"hello world\"");
        assertEquals("hello world", parser.parse("\"hello world\""));
    }

    @Test(timeout = 1000)
    public void testParseNumber() {
        SimpleJsonParser parser = new SimpleJsonParser("42");
        assertEquals(42.0, parser.parse("42"), 0.001);
    }

    @Test(timeout = 1000)
    public void testParseNegativeNumber() {
        SimpleJsonParser parser = new SimpleJsonParser("-42");
        assertEquals(-42.0, parser.parse("-42"), 0.001);
    }

    @Test(timeout = 1000)
    public void testParseFloatingPointNumber() {
        SimpleJsonParser parser = new SimpleJsonParser("3.14");
        assertEquals(3.14, parser.parse("3.14"), 0.001);
    }

    @Test(timeout = 1000)
    public void testParseBooleanTrue() {
        SimpleJsonParser parser = new SimpleJsonParser("true");
        assertTrue(parser.parse("true"));
    }

    @Test(timeout = 1000)
    public void testParseBooleanFalse() {
        SimpleJsonParser parser = new SimpleJsonParser("false");
        assertFalse(parser.parse("false"));
    }

    @Test(timeout = 1000)
    public void testParseArray() {
        SimpleJsonParser parser = new SimpleJsonParser("[1, 2, 3]");
        assertEquals(new int[]{1, 2, 3}, parser.parse("[1, 2, 3]"));
    }

    @Test(timeout = 1000)
    public void testParseNestedArray() {
        SimpleJsonParser parser = new SimpleJsonParser("[[1, 2], [3, 4]]");
        assertEquals(new int[][]{{1, 2}, {3, 4}}, parser.parse("[[1, 2], [3, 4]]"));
    }

    @Test(timeout = 1000)
    public void testParseNestedObject() {
        SimpleJsonParser parser = new SimpleJsonParser("{\"key\": \"value\"}");
        assertEquals("value", parser.parse("{\"key\": \"value\"}"));
    }
}
