package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest12 {

    @Test(timeout = 1000)
    public void testSimpleJsonParsing() {
        String input = "{\"name\":\"John\", \"age\":30}";
        SimpleJsonParser parser = new SimpleJsonParser();
        Object result = parser.parse(input);
        assertTrue(result instanceof java.util.Map);
        Map<String, Object> map = (Map<String, Object>) result;
        assertEquals("John", map.get("name"));
        assertEquals(30, map.get("age"));
    }

    @Test(timeout = 1000)
    public void testComplexJsonParsing() {
        String input = "[{\"name\":\"John\", \"age\":30}, {\"name\":\"Jane\", \"age\":25}]";
        SimpleJsonParser parser = new SimpleJsonParser();
        Object result = parser.parse(input);
        assertTrue(result instanceof List);
        List<Object> list = (List<Object>) result;
        assertEquals(2, list.size());

        Map<String, Object> firstItem = (Map<String, Object>) list.get(0);
        assertEquals("John", firstItem.get("name"));
        assertEquals(30, firstItem.get("age"));

        Map<String, Object> secondItem = (Map<String, Object>) list.get(1);
        assertEquals("Jane", secondItem.get("name"));
        assertEquals(25, secondItem.get("age"));
    }
}
