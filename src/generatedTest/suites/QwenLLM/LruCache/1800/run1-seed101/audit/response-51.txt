package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest51 {

    @Test(timeout = 1000)
    public void testEdgeCasesAndOperationSequences() {
        LruCache<Integer, String> cache = new LruCache<>(3);

        // Test put and get with single entry
        cache.put(1, "one");
        assertEquals("one", cache.get(1));

        // Test evict and reinsert
        cache.put(2, "two");
        assertEquals("one", cache.get(1));
        assertEquals("two", cache.get(2));
        cache.put(3, "three");
        assertEquals("two", cache.get(2)); // "two" should be evicted
        assertEquals("three", cache.get(3));

        // Test put after eviction and null check
        cache.put(4, "four");
        assertEquals("three", cache.get(3)); // "three" should still be there
        assertNull(cache.get(2)); // "two" has been evicted

        // Test size
        assertEquals(2, cache.size());

        // Test containsKey
        assertTrue(cache.containsKey(3));
        assertFalse(cache.containsKey(2));

        // Test get with non-existent key
        assertNull(cache.get(5));

        // Test put with null key
        try {
            cache.put(null, "nullValue");
            fail("Expected NullPointerException for null key");
        } catch (NullPointerException e) {
            // Expected exception
        }

        // Test put with null value
        cache.put(6, null);
        assertNull(cache.get(6));

        // Test clear
        cache.clear();
        assertEquals(0, cache.size());
        assertFalse(cache.containsKey(1));
        assertFalse(cache.containsKey(2));
        assertFalse(cache.containsKey(3));
        assertFalse(cache.containsKey(4));
        assertFalse(cache.containsKey(6));
    }
}
