package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest44 {

    @Test(timeout = 1000)
    public void testBasicOperations() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Put some entries
        cache.put("one", 1);
        cache.put("two", 2);
        cache.put("three", 3);

        // Check size and containsKey
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("one"));
        assertTrue(cache.containsKey("two"));
        assertTrue(cache.containsKey("three"));

        // Get an entry
        assertEquals(2, (int) cache.get("two"));

        // Put more entries to exceed capacity
        cache.put("four", 4);
        cache.put("five", 5);

        // Check size and containsKey after overflow
        assertEquals(3, cache.size());
        assertFalse(cache.containsKey("one"));
        assertTrue(cache.containsKey("two"));
        assertTrue(cache.containsKey("three"));
        assertTrue(cache.containsKey("four"));
        assertTrue(cache.containsKey("five"));
    }
}
