package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest9 {

    @Test(timeout = 1000)
    public void testLruCache() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Test adding elements and checking size
        cache.put("one", 1);
        cache.put("two", 2);
        cache.put("three", 3);
        assertEquals(3, cache.size());
        assertEquals(1, cache.get("one"));
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.get("three"));

        // Test overwriting an existing element
        cache.put("four", 4);
        assertEquals(3, cache.size());
        assertEquals(4, cache.get("four"));
        assertEquals(1, cache.get("one"));

        // Test removing an existing element
        cache.remove("one");
        assertEquals(2, cache.size());
        assertNull(cache.get("one"));

        // Test adding an element that would exceed capacity
        cache.put("five", 5);
        assertEquals(3, cache.size());
        assertEquals(5, cache.get("five"));
        assertNull(cache.get("two")); // "two" was the least recently used

        // Test containsKey
        assertTrue(cache.containsKey("five"));
        assertFalse(cache.containsKey("one"));

        // Test clear
        cache.clear();
        assertEquals(0, cache.size());
        assertNull(cache.get("five"));
    }
}
