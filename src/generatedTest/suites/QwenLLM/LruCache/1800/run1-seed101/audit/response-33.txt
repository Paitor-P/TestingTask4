package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest33 {

    @Test(timeout = 1000)
    public void testLruCacheBasics() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Put some items into the cache
        lruCache.put("A", 1);
        lruCache.put("B", 2);
        lruCache.put("C", 3);

        // Check the size of the cache
        assertEquals(3, lruCache.size());

        // Check the presence of items
        assertTrue(lruCache.containsKey("A"));
        assertTrue(lruCache.containsKey("B"));
        assertTrue(lruCache.containsKey("C"));

        // Get an item
        assertEquals(1, lruCache.get("A").intValue());

        // Put an item that evicts another item
        lruCache.put("D", 4);

        // Check the size of the cache and the presence of items
        assertEquals(3, lruCache.size());
        assertFalse(lruCache.containsKey("A"));
        assertTrue(lruCache.containsKey("B"));
        assertTrue(lruCache.containsKey("C"));
        assertTrue(lruCache.containsKey("D"));
    }
}
