package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest39 {

    @Test(timeout = 1000)
    public void testPutAndGet() {
        LruCache<String, Integer> cache = new LruCache<>(3);
        cache.put("key1", 1);
        cache.put("key2", 2);
        cache.put("key3", 3);

        assertEquals(1, (int) cache.get("key1"));
        assertEquals(2, (int) cache.get("key2"));
        assertEquals(3, (int) cache.get("key3"));

        // Put more items to exceed capacity
        cache.put("key4", 4);

        // The least recently used key ("key1") should be evicted
        assertNull(cache.get("key1"));
        assertEquals(2, (int) cache.get("key2"));
        assertEquals(3, (int) cache.get("key3"));
        assertEquals(4, (int) cache.get("key4"));
    }
}
