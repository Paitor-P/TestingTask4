package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest42 {

    @Test(timeout = 1000)
    public void testBasicOperations() {
        LruCache<Integer, String> cache = new LruCache<>(3);

        // Test basic put and get operations
        cache.put(1, "one");
        cache.put(2, "two");
        assertEquals("one", cache.get(1));
        assertEquals(2, cache.size());

        cache.put(3, "three");
        assertEquals("two", cache.get(2));
        assertEquals("three", cache.get(3));
        assertEquals(3, cache.size());

        // Test cache evicts the least recently used item
        cache.put(4, "four");
        assertEquals("three", cache.get(2));
        assertEquals(4, cache.size());

        // Test containsKey and size
        assertTrue(cache.containsKey(3));
        assertFalse(cache.containsKey(5));
        assertEquals(4, cache.size());

        // Test edge case: capacity of 1
        LruCache<Integer, String> cache1 = new LruCache<>(1);
        cache1.put(1, "one");
        cache1.put(2, "two");
        assertNull(cache1.get(1));
        assertEquals(1, cache1.size());
        cache1.put(3, "three");
        assertNull(cache1.get(2));
        assertEquals(1, cache1.size());

        // Test edge case: get non-existent key
        assertNull(cache.get(4));
        assertEquals(1, cache.size());

        // Test edge case: remove existing key
        cache.put(5, "five");
        assertEquals("five", cache.get(5));
        assertEquals(2, cache.size());
        cache.remove(5);
        assertNull(cache.get(5));
        assertEquals(1, cache.size());
    }
}
