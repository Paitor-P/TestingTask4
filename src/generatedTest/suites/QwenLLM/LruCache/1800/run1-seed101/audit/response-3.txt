package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest3 {

    @Test(timeout = 1000)
    public void testLruCache() {
        LruCache<Integer, String> lruCache = new LruCache<>(3);

        // Test basic operations
        lruCache.put(1, "one");
        lruCache.put(2, "two");
        lruCache.put(3, "three");

        assertEquals("one", lruCache.get(1));
        assertEquals("two", lruCache.get(2));
        assertEquals("three", lruCache.get(3));

        // Test capacity limit
        lruCache.put(4, "four");
        assertEquals("one", lruCache.get(1)); // "one" should be evicted
        assertEquals("two", lruCache.get(2)); // "two" should be evicted
        assertEquals("three", lruCache.get(3)); // "three" should remain

        // Test put and get
        lruCache.put(1, "one updated");
        assertEquals("one updated", lruCache.get(1));
        assertEquals("three", lruCache.get(2)); // "two" should be evicted
        assertEquals("three", lruCache.get(3)); // "three" should remain

        // Test containsKey
        assertTrue(lruCache.containsKey(1));
        assertFalse(lruCache.containsKey(4));

        // Test size
        assertEquals(2, lruCache.size());
    }
}
