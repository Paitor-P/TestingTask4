package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;
import java.util.Map;

public class QwenTest6 {

    @Test(timeout = 1000)
    public void testBasicOperation() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Test put and get
        lruCache.put("a", 1);
        assertEquals(1, lruCache.get("a"));
        assertEquals(1, lruCache.size());

        lruCache.put("b", 2);
        assertEquals(2, lruCache.get("b"));
        assertEquals(2, lruCache.size());

        lruCache.put("c", 3);
        assertEquals(3, lruCache.get("c"));
        assertEquals(3, lruCache.size());

        // Test put with capacity reached
        lruCache.put("d", 4); // "a" should be evicted
        assertNull(lruCache.get("a"));
        assertEquals(3, lruCache.size());

        // Test containsKey
        assertTrue(lruCache.containsKey("b"));
        assertFalse(lruCache.containsKey("a"));

        // Test size
        assertEquals(3, lruCache.size());

        // Test get non-existent key
        assertNull(lruCache.get("e"));

        // Test put with same key
        lruCache.put("b", 5); // "b" should be updated
        assertEquals(5, lruCache.get("b"));
        assertEquals(3, lruCache.size());

        // Test put with different key
        lruCache.put("f", 6); // "c" should be evicted
        assertNull(lruCache.get("c"));
        assertEquals(3, lruCache.size());

        // Test containsKey after evictions
        assertFalse(lruCache.containsKey("b"));
        assertTrue(lruCache.containsKey("f"));

        // Test eviction order
        lruCache.put("g", 7); // "d" should be evicted
        assertNull(lruCache.get("d"));
        assertEquals(3, lruCache.size());

        // Test get after eviction
        assertEquals(5, lruCache.get("b"));
        assertEquals(7, lruCache.get("f"));
        assertEquals(3, lruCache.size());

        // Test put after eviction
        lruCache.put("h", 8); // "g" should be evicted
        assertNull(lruCache.get("g"));
        assertEquals(3, lruCache.size());

        // Test containsKey after eviction and put
        assertFalse(lruCache.containsKey("d"));
        assertTrue(lruCache.containsKey("h"));

        // Test clear
        lruCache.clear();
        assertEquals(0, lruCache.size());
    }
}
