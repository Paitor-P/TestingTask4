package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LongestIncreasingSubsequence;

public class QwenTest35 {

    @Test(timeout=1000)
    public void testLengthOfLIS() {
        // Test case 1: Empty array
        assertEquals(0, LongestIncreasingSubsequence.lengthOfLIS(new int[]{}));

        // Test case 2: Single element array
        assertEquals(1, LongestIncreasingSubsequence.lengthOfLIS(new int[]{5}));

        // Test case 3: Ascending sequence
        assertEquals(5, LongestIncreasingSubsequence.lengthOfLIS(new int[]{1, 2, 3, 4, 5}));

        // Test case 4: Descending sequence
        assertEquals(1, LongestIncreasingSubsequence.lengthOfLIS(new int[]{5, 4, 3, 2, 1}));

        // Test case 5: Random sequence with duplicates
        assertEquals(4, LongestIncreasingSubsequence.lengthOfLIS(new int[]{1, 3, 2, 4, 5, 5, 6}));

        // Test case 6: Sequence with negative numbers
        assertEquals(3, LongestIncreasingSubsequence.lengthOfLIS(new int[]{-1, -3, -2, -4, -5, -6}));

        // Test case 7: Sequence with negative and positive numbers
        assertEquals(4, LongestIncreasingSubsequence.lengthOfLIS(new int[]{-1, 2, -3, 4, -5, 6, 7}));

        // Test case 8: Sequence with repeated elements
        assertEquals(3, LongestIncreasingSubsequence.lengthOfLIS(new int[]{1, 2, 2, 3, 4, 5, 5}));
    }
