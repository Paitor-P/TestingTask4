package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest31 {

    @Test(timeout = 1000)
    public void testLengthOfLIS() {
        // Test case 1: Normal input
        int[] numbers1 = {10, 9, 2, 5, 3, 7, 101, 18};
        assertEquals(4, new LongestIncreasingSubsequence().lengthOfLIS(numbers1));

        // Test case 2: All elements are the same
        int[] numbers2 = {1, 1, 1, 1, 1};
        assertEquals(1, new LongestIncreasingSubsequence().lengthOfLIS(numbers2));

        // Test case 3: Empty array
        int[] numbers3 = {};
        assertEquals(0, new LongestIncreasingSubsequence().lengthOfLIS(numbers3));

        // Test case 4: Single element
        int[] numbers4 = {5};
        assertEquals(1, new LongestIncreasingSubsequence().lengthOfLIS(numbers4));

        // Test case 5: Decreasing sequence
        int[] numbers5 = {9, 7, 5, 3, 1};
        assertEquals(1, new LongestIncreasingSubsequence().lengthOfLIS(numbers5));

        // Test case 6: Increasing sequence
        int[] numbers6 = {1, 2, 3, 4, 5};
        assertEquals(5, new LongestIncreasingSubsequence().lengthOfLIS(numbers6));

        // Test case 7: Random sequence
        int[] numbers7 = {1, 3, 2, 5, 4, 8, 7, 6};
        assertEquals(4, new LongestIncreasingSubsequence().lengthOfLIS(numbers7));

        // Test case 8: Repeated elements in the middle
        int[] numbers8 = {1, 3, 2, 5, 5, 4, 8, 7, 6};
        assertEquals(4, new LongestIncreasingSubsequence().lengthOfLIS(numbers8));
    }
}
