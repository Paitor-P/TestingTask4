package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest1 {

    @Test(timeout = 1000)
    public void testPutGetContainsKey() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Put some values into the cache
        cache.put("a", 1);
        cache.put("b", 2);
        cache.put("c", 3);

        // Check that the values are present and correctly retrieved
        assertEquals(1, cache.get("a"));
        assertEquals(2, cache.get("b"));
        assertEquals(3, cache.get("c"));

        // Check that the keys are present
        assertTrue(cache.containsKey("a"));
        assertTrue(cache.containsKey("b"));
        assertTrue(cache.containsKey("c"));

        // Check that the size is correct
        assertEquals(3, cache.size());

        // Put more values to test eviction
        cache.put("d", 4);
        cache.put("e", 5);

        // Check that the least recently used key ("a") is evicted
        assertNull(cache.get("a"));
        assertTrue(cache.containsKey("b"));
        assertTrue(cache.containsKey("c"));
        assertTrue(cache.containsKey("d"));
        assertTrue(cache.containsKey("e"));

        // Check that the size is updated
        assertEquals(3, cache.size());
    }
}
