package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest12 {

    @Test(timeout = 1000)
    public void testLruCacheWithMultiplePutOperationsAndSize() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Add three elements
        lruCache.put("one", 1);
        lruCache.put("two", 2);
        lruCache.put("three", 3);

        // Verify size is 3
        assertEquals(3, lruCache.size());

        // Add another element to exceed capacity
        lruCache.put("four", 4);

        // Verify "one" is removed and "four" is added, size is 3
        assertEquals(3, lruCache.size());
        assertTrue(lruCache.containsKey("two"));
        assertFalse(lruCache.containsKey("one"));
        assertTrue(lruCache.containsKey("four"));
    }
}
