package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest28 {

    @Test(timeout = 1000)
    public void testLRUCache() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Add some entries to the cache
        lruCache.put("A", 1);
        lruCache.put("B", 2);
        lruCache.put("C", 3);

        // Check the size of the cache
        assertEquals(3, lruCache.size());

        // Check the value of an existing key
        assertEquals(2, lruCache.get("B").intValue());

        // Add an existing key with a new value
        lruCache.put("B", 4);

        // The cache should still have 3 entries
        assertEquals(3, lruCache.size());

        // Check the value of the updated key
        assertEquals(4, lruCache.get("B").intValue());

        // Add a new key to the cache
        lruCache.put("D", 5);

        // The cache should have 3 entries, with "B" being the least recently used
        assertEquals(3, lruCache.size());

        // Check the value of the removed key
        assertNull(lruCache.get("A"));

        // The cache should have 2 entries, with "B" and "C" being the most recently used
        assertEquals(2, lruCache.size());
    }
}
