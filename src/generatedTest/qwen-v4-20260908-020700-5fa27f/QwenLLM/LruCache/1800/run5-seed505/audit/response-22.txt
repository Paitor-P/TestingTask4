package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import com.viktor.lab4.LruCache;
import java.util.Map;

public class QwenTest22 {

    @Test(timeout = 1000)
    public void testSimplePutGet() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        cache.put("one", 1);
        cache.put("two", 2);
        cache.put("three", 3);

        assertEquals(3, cache.size());

        assertEquals(1, cache.get("one"));
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.get("three"));

        // Simulate a read that should not cause a removal
        cache.get("two");

        // Ensure the order is maintained with LRU
        assertEquals(1, cache.get("one"));
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.get("three"));

        // Put a new key to exceed capacity
        cache.put("four", 4);

        // Check that the oldest key was removed
        assertNull(cache.get("one"));
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.get("three"));
        assertEquals(4, cache.get("four"));
    }
}
