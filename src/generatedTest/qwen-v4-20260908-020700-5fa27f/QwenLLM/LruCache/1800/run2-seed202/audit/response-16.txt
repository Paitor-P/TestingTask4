package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest16 {

    @Test(timeout = 1000)
    public void testBasicGetPut() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Put some entries into the cache
        cache.put("one", 1);
        cache.put("two", 2);
        cache.put("three", 3);

        // Check if the cache size is correct
        assertEquals(3, cache.size());

        // Check if the entries are correctly retrieved
        assertEquals(1, cache.get("one"));
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.get("three"));

        // Put another entry to evict the least recently used entry
        cache.put("four", 4);

        // Check if the least recently used entry "one" is removed
        assertNull(cache.get("one"));
        assertEquals(2, cache.size());
    }
}
