package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.Map;

public class QwenTest8 {

    @Test(timeout = 1000)
    public void testEmptyCache() {
        // Create an empty LruCache with a capacity of 1
        LruCache<String, String> cache = new LruCache<>(1);

        // Check that the cache size is 0
        assertEquals(0, cache.size());

        // Check that the cache contains no keys
        assertFalse(cache.containsKey("key"));

        // Check that the cache returns null for non-existent keys
        assertNull(cache.get("key"));

        // Check that putting a key-value pair does not affect the size
        cache.put("key", "value");
        assertEquals(0, cache.size());

        // Check that putting a key-value pair does not affect the containsKey result
        assertFalse(cache.containsKey("key"));
    }
}
