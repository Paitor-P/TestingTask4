package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest12 {

    @Test(timeout = 1000)
    public void testRemoveEldestEntryOnThreshold() {
        // Create a LruCache with a capacity of 3
        LruCache<Integer, String> cache = new LruCache<>(3);

        // Put three entries into the cache
        cache.put(1, "A");
        cache.put(2, "B");
        cache.put(3, "C");

        // Verify that size is 3
        assertEquals(3, cache.size());

        // Put a fourth entry, which should evict the oldest entry (key 1)
        cache.put(4, "D");

        // Verify that the size is now 3 again, as the eldest entry was removed
        assertEquals(3, cache.size());

        // Check that key 1 is no longer in the cache
        assertFalse(cache.containsKey(1));

        // Check that the size is 3 after removing the eldest entry
        assertEquals(3, cache.size());

        // Verify the order of entries in the cache
        // LinkedHashMap maintains the insertion order, so the last entry should be key 4
        assertTrue(cache.containsKey(4));
        assertFalse(cache.containsKey(2));
        assertFalse(cache.containsKey(3));
    }
}
