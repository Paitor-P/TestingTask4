package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest53 {

    @Test(timeout = 1000)
    public void testLruCacheBasicOperations() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Test put and get operations
        lruCache.put("one", 1);
        lruCache.put("two", 2);
        lruCache.put("three", 3);
        assertEquals(3, lruCache.size());
        assertEquals(1, (int) lruCache.get("one"));
        assertEquals(2, (int) lruCache.get("two"));
        assertEquals(3, (int) lruCache.get("three"));

        // Test put and get with capacity reached
        lruCache.put("four", 4);
        assertEquals(3, lruCache.size());
        assertEquals(1, (int) lruCache.get("one"));
        assertEquals(2, (int) lruCache.get("two"));
        assertEquals(4, (int) lruCache.get("four"));

        // Test put and get with new key
        lruCache.put("five", 5);
        assertEquals(3, lruCache.size());
        assertEquals(2, (int) lruCache.get("two"));
        assertEquals(4, (int) lruCache.get("four"));
        assertEquals(5, (int) lruCache.get("five"));

        // Test get and containsKey
        assertEquals(4, (int) lruCache.get("four"));
        assertTrue(lruCache.containsKey("four"));
        assertFalse(lruCache.containsKey("six"));
    }
}
