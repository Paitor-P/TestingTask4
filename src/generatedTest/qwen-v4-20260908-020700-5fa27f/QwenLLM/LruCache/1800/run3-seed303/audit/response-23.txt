package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest23 {

    @Test(timeout=1000)
    public void testPutAndGet() {
        LruCache<Integer, String> cache = new LruCache<>(3);
        cache.put(1, "one");
        cache.put(2, "two");
        cache.put(3, "three");

        // Check initial size and content
        assertEquals(3, cache.size());
        assertEquals("one", cache.get(1));
        assertEquals("two", cache.get(2));
        assertEquals("three", cache.get(3));

        // Add a new key to evict the least recently used key (1)
        cache.put(4, "four");

        // Check size and content after eviction
        assertEquals(3, cache.size());
        assertNull(cache.get(1));
        assertEquals("two", cache.get(2));
        assertEquals("four", cache.get(4)); // Updated expected value

        // Ensure other entries remain unchanged
        assertEquals("one", cache.get(2));
        assertEquals("three", cache.get(3));
    }
}
