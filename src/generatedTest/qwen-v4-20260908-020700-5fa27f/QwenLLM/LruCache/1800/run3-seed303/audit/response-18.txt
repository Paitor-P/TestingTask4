package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest18 {

    @Test(timeout = 1000)
    public void testCombinedEffectOfGetAndPut() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Add initial items
        cache.put("key1", 1);
        cache.put("key2", 2);
        cache.put("key3", 3);

        // Check size and presence of items
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("key1"));
        assertTrue(cache.containsKey("key2"));
        assertTrue(cache.containsKey("key3"));

        // Get and put an existing key
        Integer value1 = cache.get("key1");
        cache.put("key1", value1 * 2);

        // Check size and presence of items, updated value
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("key1"));
        assertEquals(2, cache.get("key1"));

        // Get and put a new key
        cache.get("key4"); // Evicts "key2" due to LRU
        cache.put("key4", 4);

        // Check size and presence of items, new key added
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("key1"));
        assertFalse(cache.containsKey("key2"));
        assertTrue(cache.containsKey("key3"));
        assertTrue(cache.containsKey("key4"));
    }
}
