package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest27 {

    @Test(timeout=1000)
    public void testLruCacheBasicOperations() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);
        lruCache.put("a", 1);
        lruCache.put("b", 2);
        lruCache.put("c", 3);
        assertEquals(3, lruCache.size());
        assertTrue(lruCache.containsKey("a"));
        assertTrue(lruCache.containsKey("b"));
        assertTrue(lruCache.containsKey("c"));
        assertNull(lruCache.get("d"));
        lruCache.put("d", 4);
        assertEquals(3, lruCache.size());
        assertTrue(lruCache.containsKey("a"));
        assertFalse(lruCache.containsKey("b"));
        assertTrue(lruCache.containsKey("c"));
        assertTrue(lruCache.containsKey("d"));
        assertEquals(4, lruCache.get("d"));
        lruCache.put("e", 5);
        assertEquals(3, lruCache.size());
        assertFalse(lruCache.containsKey("a"));
        assertTrue(lruCache.containsKey("b"));
        assertTrue(lruCache.containsKey("c"));
        assertFalse(lruCache.containsKey("d"));
        assertTrue(lruCache.containsKey("e"));
    }
}
