package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest4 {

    @Test(timeout = 1000)
    public void testBasicLruCacheOperations() {
        LruCache<String, Integer> lruCache = new LruCache<>(2);

        // Put some initial items
        lruCache.put("A", 1);
        lruCache.put("B", 2);

        // Check the size
        assertEquals(2, lruCache.size());

        // Check the get operation
        assertEquals(1, lruCache.get("A"));
        assertEquals(2, lruCache.get("B"));

        // Check the containsKey operation
        assertTrue(lruCache.containsKey("A"));
        assertFalse(lruCache.containsKey("C"));

        // Add an item that evicts "A"
        lruCache.put("C", 3);

        // Check that "A" is evicted and "C" is added
        assertEquals(2, lruCache.size());
        assertNull(lruCache.get("A"));
        assertEquals(3, lruCache.get("C"));

        // Put more items to exceed capacity
        lruCache.put("D", 4);
        lruCache.put("E", 5);

        // Check that "B" is evicted
        assertEquals(2, lruCache.size());
        assertNull(lruCache.get("B"));
        assertEquals(3, lruCache.get("C"));
    }
}
