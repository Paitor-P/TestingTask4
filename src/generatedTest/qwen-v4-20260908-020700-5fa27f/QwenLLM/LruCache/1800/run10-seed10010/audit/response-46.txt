package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.Map;

public class QwenTest46 {

    @Test(timeout = 1000)
    public void testLruCachePutAndGet() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Put elements into the cache
        lruCache.put("one", 1);
        lruCache.put("two", 2);
        lruCache.put("three", 3);

        // Check that all elements are present and correct
        assertEquals(3, lruCache.size());
        assertEquals(1, lruCache.get("one"));
        assertEquals(2, lruCache.get("two"));
        assertEquals(3, lruCache.get("three"));

        // Put a new element, which will evict the least recently used element
        lruCache.put("four", 4);

        // Check that "one" has been evicted
        assertNull(lruCache.get("one"));
        assertEquals(3, lruCache.size());

        // Put another new element, which will evict "two"
        lruCache.put("five", 5);

        // Check that "two" has been evicted
        assertNull(lruCache.get("two"));
        assertEquals(2, lruCache.size());

        // Check that "three" is still present and correct
        assertEquals(3, lruCache.get("three"));
        assertEquals(2, lruCache.size());

        // Check that "four" is still present and correct
        assertEquals(4, lruCache.get("four"));
        assertEquals(2, lruCache.size());

        // Check that "five" is now the only element in the cache
        assertEquals(1, lruCache.size());
        assertEquals(5, lruCache.get("five"));
    }
}
