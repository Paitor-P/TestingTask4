package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest50 {

    @Test(timeout = 1000)
    public void testBasicOperations() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Put operations
        lruCache.put("A", 1);
        lruCache.put("B", 2);
        lruCache.put("C", 3);

        // Check size and content
        assertEquals(3, lruCache.size());
        assertTrue(lruCache.containsKey("A"));
        assertTrue(lruCache.containsKey("B"));
        assertTrue(lruCache.containsKey("C"));

        // Get operations
        assertEquals(1, lruCache.get("A"));
        assertEquals(2, lruCache.get("B"));
        assertEquals(3, lruCache.get("C"));

        // Check after some operations
        lruCache.put("D", 4); // "A" is least recently used, should be evicted
        assertFalse(lruCache.containsKey("A"));
        assertEquals(3, lruCache.size());
    }
}
