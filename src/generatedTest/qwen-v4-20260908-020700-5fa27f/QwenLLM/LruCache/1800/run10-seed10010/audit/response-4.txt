package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest4 {

    @Test(timeout = 1000)
    public void testZeroCapacityCache() {
        // Create an LruCache with capacity 0
        LruCache<String, Integer> cache = new LruCache<>(0);

        // Check that the cache is empty
        assertTrue(cache.isEmpty());
        assertEquals(0, cache.size());

        // Attempt to put a key-value pair
        cache.put("key1", 1);

        // Check that the cache is empty after a put operation
        assertTrue(cache.isEmpty());
        assertEquals(0, cache.size());
    }
}
