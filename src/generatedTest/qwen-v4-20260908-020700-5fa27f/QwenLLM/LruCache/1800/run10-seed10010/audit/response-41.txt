package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest41 {

    @Test(timeout = 1000)
    public void testPutAndGet() {
        LruCache<Integer, String> lruCache = new LruCache<>(3);

        // Put initial values
        lruCache.put(1, "one");
        lruCache.put(2, "two");
        lruCache.put(3, "three");

        // Check initial size
        assertEquals(3, lruCache.size());

        // Get existing values
        assertEquals("one", lruCache.get(1));
        assertEquals("two", lruCache.get(2));
        assertEquals("three", lruCache.get(3));

        // Check size after getting
        assertEquals(3, lruCache.size());

        // Put a new value, evicting the oldest
        lruCache.put(4, "four");

        // Check size after eviction
        assertEquals(3, lruCache.size());

        // Check that the oldest key was evicted
        assertNull(lruCache.get(1));
    }
}
