package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest25 {

    @Test(timeout = 1000)
    public void testBasicUsage() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Put some values into the cache
        cache.put("A", 1);
        cache.put("B", 2);
        cache.put("C", 3);

        // Check that the cache size is 3
        assertEquals(3, cache.size());

        // Access an existing value
        assertEquals(2, cache.get("B"));

        // Check that the cache size is still 3 (no change)
        assertEquals(3, cache.size());

        // Access a non-existing value
        assertNull(cache.get("D"));

        // Check that the cache size remains 3
        assertEquals(3, cache.size());

        // Put a new value to evict the least recently used item
        cache.put("E", 4);

        // Check that the cache size is 3 and "A" is no longer in the cache
        assertEquals(3, cache.size());
        assertFalse(cache.containsKey("A"));

        // Check that "E" is in the cache
        assertTrue(cache.containsKey("E"));
    }
}
