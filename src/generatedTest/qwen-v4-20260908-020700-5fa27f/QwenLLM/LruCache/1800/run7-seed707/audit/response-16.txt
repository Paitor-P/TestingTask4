package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest16 {

    @Test(timeout = 1000)
    public void testSimplePutAndGet() {
        LruCache<Integer, String> cache = new LruCache<>(2);
        cache.put(1, "A");
        cache.put(2, "B");
        assertEquals("B", cache.get(2)); // This should pass
        assertEquals("A", cache.get(1)); // This should pass
        assertNull(cache.get(3)); // This should pass
        cache.put(3, "C"); // Now 2 is least recently used, 3 is most recently used
        assertEquals("C", cache.get(3)); // This should pass
        assertEquals("B", cache.get(2)); // This should pass
        assertNull(cache.get(1)); // This should pass
    }
}
