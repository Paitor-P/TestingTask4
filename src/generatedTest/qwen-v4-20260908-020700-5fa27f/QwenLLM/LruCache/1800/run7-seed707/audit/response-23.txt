package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

import java.util.Map;

public class QwenTest23 {

    @Test(timeout = 1000)
    public void testRemoveEldestEntryWhenCapacityIsFull() {
        LruCache<Integer, String> cache = new LruCache<>(3);

        // Add three entries
        cache.put(1, "One");
        cache.put(2, "Two");
        cache.put(3, "Three");

        // Check the size is exactly the capacity
        assertEquals(3, cache.size());

        // Add a fourth entry, which should evict the least recently used entry (1)
        cache.put(4, "Four");

        // Verify that the least recently used entry (1) is removed
        assertFalse(cache.containsKey(1));
        assertEquals(3, cache.size());
    }
}
