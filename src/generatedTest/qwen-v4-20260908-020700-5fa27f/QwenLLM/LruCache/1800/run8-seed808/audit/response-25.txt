package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import com.viktor.lab4.LruCache;

import java.util.Map;

public class QwenTest25 {

    @Test(timeout = 1000)
    public void testNormalCase() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);

        // Put some items into the cache
        lruCache.put("one", 1);
        lruCache.put("two", 2);
        lruCache.put("three", 3);

        // Check size
        assertEquals(3, lruCache.size());

        // Get an item
        assertEquals(2, lruCache.get("two"));

        // Put an existing item with a new value
        lruCache.put("two", 22);

        // Check the updated value
        assertEquals(22, lruCache.get("two"));

        // Check size after put
        assertEquals(3, lruCache.size());

        // Put a new item that will evict "one"
        lruCache.put("four", 4);

        // Check size after eviction
        assertEquals(3, lruCache.size());

        // Check that "one" is no longer in the cache
        assertFalse(lruCache.containsKey("one"));
    }
}
