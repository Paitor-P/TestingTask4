package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest43 {

    @Test(timeout = 1000)
    public void testPutAndGetSize() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);
        lruCache.put("one", 1);
        lruCache.put("two", 2);
        lruCache.put("three", 3);
        lruCache.put("four", 4); // This should evict "one"

        assertEquals(3, lruCache.size());
        assertTrue(lruCache.containsKey("two"));
        assertTrue(lruCache.containsKey("three"));
        assertFalse(lruCache.containsKey("one"));
    }
}
