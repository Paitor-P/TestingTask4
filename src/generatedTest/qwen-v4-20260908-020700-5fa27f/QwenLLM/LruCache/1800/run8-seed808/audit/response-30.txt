package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest30 {

    @Test(timeout = 1000)
    public void testPutAndGet() {
        LruCache<String, Integer> cache = new LruCache<>(3);

        // Add some elements to the cache
        cache.put("A", 1);
        cache.put("B", 2);
        cache.put("C", 3);

        // Assert the size of the cache
        assertEquals(3, cache.size());

        // Assert that the cache contains the elements
        assertTrue(cache.containsKey("A"));
        assertTrue(cache.containsKey("B"));
        assertTrue(cache.containsKey("C"));

        // Get an element and assert its value
        assertEquals(2, cache.get("B"));

        // Add a new element to the cache, which should evict the least recently used element
        cache.put("D", 4);

        // Assert that the cache size is still 3
        assertEquals(3, cache.size());

        // Assert that the least recently used element ("A") has been evicted
        assertFalse(cache.containsKey("A"));
    }
}
