package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest6 {

    @Test(timeout = 1000)
    public void testEmptyCache() {
        // Create an empty LruCache
        LruCache<String, String> cache = new LruCache<>(1);

        // Perform operations on the cache
        assertNull(cache.get("key")); // Get non-existent key
        assertFalse(cache.containsKey("key")); // Check for non-existent key
        assertEquals(0, cache.size()); // Check size of cache

        // Put an element
        cache.put("key", "value");

        // Check if the element was added correctly
        assertEquals("value", cache.get("key"));
        assertTrue(cache.containsKey("key"));
        assertEquals(1, cache.size());

        // Remove the element
        cache.remove("key"); // Corrected method name

        // Check if the element was removed correctly
        assertNull(cache.get("key"));
        assertFalse(cache.containsKey("key"));
        assertEquals(0, cache.size());
    }
}
