package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest1 {

    @Test(timeout = 1000)
    public void testPutAndGet() {
        LruCache<String, Integer> cache = new LruCache<>(3);
        cache.put("one", 1);
        assertEquals(1, cache.get("one"));
        assertEquals(3, cache.size());
        cache.put("two", 2);
        assertEquals(2, cache.get("two"));
        assertEquals(3, cache.size());
        cache.put("three", 3);
        assertEquals(3, cache.size());
        assertEquals(3, cache.get("three"));
        assertEquals(3, cache.size());
        cache.put("four", 4); // This should evict "one"
        assertEquals(4, cache.get("four"));
        assertEquals(3, cache.size());
        assertEquals(false, cache.containsKey("one"));
        assertEquals(true, cache.containsKey("two"));
        assertEquals(true, cache.containsKey("three"));
        assertEquals(true, cache.containsKey("four"));
    }
}
