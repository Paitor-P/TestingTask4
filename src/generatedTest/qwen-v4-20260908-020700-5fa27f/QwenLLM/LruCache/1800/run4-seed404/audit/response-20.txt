package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.Map;

public class QwenTest20 {

    @Test(timeout = 1000)
    public void testGetAndPut() {
        LruCache<String, Integer> lruCache = new LruCache<>(3);
        
        // Put some items into the cache
        lruCache.put("A", 1);
        lruCache.put("B", 2);
        lruCache.put("C", 3);
        
        // Check the size of the cache
        assertEquals(3, lruCache.size());
        
        // Get an item from the cache
        Integer value = lruCache.get("B");
        assertNotNull(value);
        assertEquals(2, (int) value);
        
        // Check the size of the cache after get
        assertEquals(3, lruCache.size());
        
        // Put another item to the cache
        lruCache.put("D", 4);
        
        // Check the size of the cache after put
        assertEquals(3, lruCache.size());
        
        // Check that the least recently used item "A" is removed
        assertNull(lruCache.get("A"));
    }
}
