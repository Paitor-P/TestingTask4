package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.LinkedHashMap;
import java.util.Map;

public class QwenTest32 {

    @Test(timeout = 1000)
    public void testPutAndGetAfterExceedingCapacity() {
        LruCache<Integer, String> cache = new LruCache<>(3);
        cache.put(1, "one");
        cache.put(2, "two");
        cache.put(3, "three");

        // Check that the cache size is equal to its capacity
        assertEquals(3, cache.size());

        // Check that the cache contains the first two entries
        assertTrue(cache.containsKey(1));
        assertTrue(cache.containsKey(2));

        // Attempt to add a new entry, which should evict the least recently used entry (key 1)
        cache.put(4, "four");

        // Check that the cache size is now one less than its capacity
        assertEquals(2, cache.size());

        // Check that the cache contains the new entry and the evicted entry is no longer present
        assertTrue(cache.containsKey(2));
        assertFalse(cache.containsKey(1));
    }
}
