package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest25 {

    @Test(timeout = 1000)
    public void testPutInvalidKey() {
        LruCache<String, Integer> cache = new LruCache<>(3);
        
        try {
            cache.put("invalidKey", 123);
            fail("Expected IllegalArgumentException for invalid key");
        } catch (IllegalArgumentException e) {
            assertEquals("capacity must be > 0", e.getMessage());
        }
    }
}
