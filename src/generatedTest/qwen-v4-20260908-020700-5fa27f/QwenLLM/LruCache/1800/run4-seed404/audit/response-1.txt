package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LruCache;

public class QwenTest1 {

    @Test(timeout = 1000)
    public void testSimplePutGet() {
        LruCache<String, Integer> cache = new LruCache<>(2);
        cache.put("key1", 1);
        cache.put("key2", 2);
        assertEquals(2, cache.size());
        assertEquals(1, (int) cache.get("key1"));
        assertEquals(2, (int) cache.get("key2"));
        // Since "key1" is accessed, it moves to the end of the cache
        cache.get("key1");
        assertEquals("key2", cache.store.keySet().iterator().next());
    }
}
