        cache.put("B", 2);
        cache.put("C", 3);
        
        assertEquals(1, cache.get("A"), 0.0);
        assertEquals(2, cache.get("B"), 0.0);
        assertEquals(3, cache.get("C"), 0.0);
        
        cache.put("D", 4); // This will evict "A"
        
        assertNull(cache.get("A"), 0.0);
        assertEquals(2, cache.get("B"), 0.0);
        assertEquals(3, cache.get("C"), 0.0);
        assertEquals(4, cache.get("D"), 0.0);
    }
}
