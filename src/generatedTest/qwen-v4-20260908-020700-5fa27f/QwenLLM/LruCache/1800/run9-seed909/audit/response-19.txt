package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest19 {

    @Test(timeout = 1000)
    public void testGetAndPut() {
        LruCache<String, Integer> cache = new LruCache<>(3);
        
        // Put three items into the cache
        cache.put("a", 1);
        cache.put("b", 2);
        cache.put("c", 3);
        
        // Check the size and containsKey method
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("a"));
        assertTrue(cache.containsKey("b"));
        assertTrue(cache.containsKey("c"));
        
        // Get an item from the cache
        int value = cache.get("a");
        
        // Check the get method and containsKey method
        assertEquals(1, value);
        assertEquals(3, cache.size());
        assertTrue(cache.containsKey("a"));
        assertTrue(cache.containsKey("b"));
        assertTrue(cache.containsKey("c"));
        
        // Put another item into the cache, evicting the least recently used item
        cache.put("d", 4);
        
        // Check the size and containsKey method
        assertEquals(3, cache.size());
        assertFalse(cache.containsKey("a"));
        assertTrue(cache.containsKey("b"));
        assertTrue(cache.containsKey("c"));
        assertTrue(cache.containsKey("d"));
    }
}
