To fix the compilation error and ensure the test runs correctly, we need to make the `calculateFinalPrice` method static. This is because JUnit tests are executed in a static context, and non-static methods cannot be accessed from static ones without creating an instance of the class.

package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest38 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceWithDiscount() {
        PricingRequest request = new PricingRequest(
                100.0,
                10,
                CustomerType.REGULAR,
                false,
                "SAVE10",
                false,
                0
        );
        double expected = 100.0 * (1.0 - 0.10) - 5.0 + 7.99;
        assertEquals(98.99, PricingEngine.calculateFinalPrice(request), 0.01);
    }

    public static double calculateFinalPrice(PricingRequest request) {
        // Original implementation here
        if (request == null) {
            throw new IllegalArgumentException("request must not be null");
        }
        if (request.basePrice() < 0.0) {
            throw new IllegalArgumentException("basePrice must be >= 0");
        }
        if (request.quantity() <= 0) {
            throw new IllegalArgumentException("quantity must be > 0");
        }
        if (request.customerType() == null) {
            throw new IllegalArgumentException("customerType must not be null");
        }
        if (request.loyaltyPoints() < 0) {
            throw new IllegalArgumentException("loyaltyPoints must be >= 0");
        }

        double subtotal = request.basePrice() * request.quantity();
        double discountPercent = 0.0;
        double flatDiscount = 0.0;
        double shipping = 7.99;

        if (request.quantity() >= 10) {
            discountPercent += 0.10;
        }
        if (request.customerType() == CustomerType.VIP) {
            discountPercent += 0.12;
        }
        if (request.seasonalSale()) {
            discountPercent += 0.15;
        }
        if (request.firstOrder()) {
            discountPercent += 0.05;
        }

        if (request.couponCode() != null && !request.couponCode().isBlank()) {
            String normalized = request.couponCode().trim().toUpperCase();
            if ("SAVE10".equals(normalized) && subtotal >= 50.0) {
                discountPercent += 0.10;
            } else if ("FLAT5".equals(normalized) && subtotal >= 20.0) {
                flatDiscount += 5.0;
            } else if ("FREESHIP".equals(normalized)) {
                shipping = 0.0;
            }
        }

        int loyaltyTier = request.loyaltyPoints() / 100;
        discountPercent += Math.min(loyaltyTier, 5) / 100.0;

        if (request.customerType() == CustomerType.VIP
                && request.quantity() >= 20
                && request.couponCode() != null
                && request.couponCode().trim().equalsIgnoreCase("SAVE10")) {
            discountPercent += 0.03;
        }

        discountPercent = Math.min(discountPercent, 0.35);

        double discountedSubtotal = subtotal * (1.0 - discountPercent);
        double total = discountedSubtotal - flatDiscount + shipping;

        if (total < 0.0) {
            total = 0.0;
        }

        return Math.round(total * 100.0) / 100.0;
    }
}
