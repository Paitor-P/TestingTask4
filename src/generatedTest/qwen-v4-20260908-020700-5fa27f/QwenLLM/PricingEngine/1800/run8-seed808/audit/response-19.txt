package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest19 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceWithDiscount() {
        PricingEngine pricingEngine = new PricingEngine(); // Create an instance of PricingEngine
        PricingRequest request = new PricingRequest(
                10.0, // basePrice
                5,    // quantity
                CustomerType.REGULAR, // customerType
                false,   // seasonalSale
                "SAVE10", // couponCode
                true,    // firstOrder
                50    // loyaltyPoints
        );

        double expected = 10.0 * 5 * (1.0 - 0.10) - 5.0 + 7.99; // 10 * 5 * 0.90 - 5 + 7.99
        assertEquals(expected, pricingEngine.calculateFinalPrice(request), 0.0001);
    }
}
