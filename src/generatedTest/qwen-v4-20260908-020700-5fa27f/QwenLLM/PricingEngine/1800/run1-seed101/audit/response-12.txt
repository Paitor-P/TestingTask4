package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest11 {

    @Test(timeout = 1000)
    public void testCalculateFinalPrice() {
        PricingEngine engine = new PricingEngine();

        PricingRequest request = new PricingRequest(
                100.0, // basePrice
                15,    // quantity
                PricingEngine.CustomerType.REGULAR,
                true,    // seasonalSale
                "SAVE10", // couponCode
                true,    // firstOrder
                500     // loyaltyPoints
        );

        double finalPrice = engine.calculateFinalPrice(request);

        // Expected result based on the given source
        double expected = 100.0 * 15 * (1.0 - 0.25) + 7.99 - 5.0; // 1127.99

        assertEquals(expected, finalPrice, 0.01);
    }
}
