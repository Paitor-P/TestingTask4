package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest11 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceVipWithLoyalty() {
        PricingRequest request = new PricingRequest(
                100.0, // basePrice
                20, // quantity
                PricingEngine.CustomerType.VIP, // customerType
                true, // seasonalSale
                "SAVE10", // couponCode
                true, // firstOrder
                1000 // loyaltyPoints
        );

        double expected = 100.0 * (1.0 - (0.10 + 0.12 + 0.15 + 0.05 + 0.10 + 0.03)) + 7.99;
        double actual = PricingEngine.calculateFinalPrice(request);

        assertEquals("Expected discount is incorrect", expected, actual, 0.001);
    }
}
