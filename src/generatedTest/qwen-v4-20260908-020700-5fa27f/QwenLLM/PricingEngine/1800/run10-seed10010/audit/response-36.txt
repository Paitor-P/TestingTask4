package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest36 {

    public static class PricingEngineTest {

        @Test(timeout = 1000)
        public void testCalculateFinalPrice() {
            PricingEngine pricingEngine = new PricingEngine();

            PricingRequest request1 = new PricingRequest(100.0, 5, CustomerType.REGULAR, false, "SAVE10", false, 100);
            assertEquals(85.0, pricingEngine.calculateFinalPrice(request1), 0.01);

            PricingRequest request2 = new PricingRequest(100.0, 5, CustomerType.VIP, false, "SAVE10", true, 100);
            assertEquals(79.25, pricingEngine.calculateFinalPrice(request2), 0.01);

            PricingRequest request3 = new PricingRequest(100.0, 10, CustomerType.REGULAR, true, null, false, 0);
            assertEquals(89.0, pricingEngine.calculateFinalPrice(request3), 0.01);

            PricingRequest request4 = new PricingRequest(100.0, 10, CustomerType.VIP, true, "FLAT5", false, 50);
            assertEquals(75.0, pricingEngine.calculateFinalPrice(request4), 0.01);

            PricingRequest request5 = new PricingRequest(100.0, 10, CustomerType.REGULAR, true, "FREESHIP", false, 0);
            assertEquals(100.0, pricingEngine.calculateFinalPrice(request5), 0.01);

            PricingRequest request6 = new PricingRequest(100.0, 20, CustomerType.VIP, true, "SAVE10", false, 100);
            assertEquals(77.1, pricingEngine.calculateFinalPrice(request6), 0.01);
        }
    }
}
