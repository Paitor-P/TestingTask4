package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest6 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceWithInvalidInput() {
        PricingEngine pricingEngine = new PricingEngine(); // Construct an instance of PricingEngine
        PricingRequest request = new PricingRequest(
                -1.0, // basePrice
                10,   // quantity
                PricingEngine.CustomerType.REGULAR,
                true,  // seasonalSale
                "SAVE10", // couponCode
                true,  // firstOrder
                -100  // loyaltyPoints
        );

        try {
            pricingEngine.calculateFinalPrice(request);
            fail("Expected IllegalArgumentException for null basePrice");
        } catch (IllegalArgumentException e) {
            assertEquals("basePrice must be >= 0", e.getMessage());
        }
    }
}
