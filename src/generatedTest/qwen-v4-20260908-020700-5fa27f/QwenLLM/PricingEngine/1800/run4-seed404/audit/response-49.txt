package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.*;

public class QwenTest49 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceEdgeCases() {
        PricingEngine engine = new PricingEngine();

        // Edge case: empty request
        PricingRequest emptyRequest = new PricingRequest(0.0, 0, CustomerType.REGULAR, false, null, false, 0);
        assertEquals(0.0, engine.calculateFinalPrice(emptyRequest), 0.001);

        // Edge case: zero quantity
        PricingRequest zeroQuantityRequest = new PricingRequest(100.0, 0, CustomerType.REGULAR, false, null, false, 0);
        assertEquals(0.0, engine.calculateFinalPrice(zeroQuantityRequest), 0.001);

        // Edge case: smallest legal input
        PricingRequest smallestLegalRequest = new PricingRequest(0.01, 1, CustomerType.REGULAR, false, null, false, 0);
        assertEquals(0.01, engine.calculateFinalPrice(smallestLegalRequest), 0.001);

        // Edge case: invalid quantity, should throw IllegalArgumentException
        PricingRequest invalidQuantityRequest = new PricingRequest(100.0, -1, CustomerType.REGULAR, false, null, false, 0);
        try {
            engine.calculateFinalPrice(invalidQuantityRequest);
            fail("Expected IllegalArgumentException for negative quantity");
        } catch (IllegalArgumentException e) {
            // Expected exception, do nothing
        }
    }
}
