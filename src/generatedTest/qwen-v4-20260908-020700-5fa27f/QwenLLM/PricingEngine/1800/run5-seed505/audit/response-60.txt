package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest60 {

    @Test(timeout = 1000)
    public void testCalculateFinalPriceWithMultipleDiscounts() {
        PricingEngine engine = new PricingEngine();
        PricingRequest request = new PricingRequest(
                100.0, 5, PricingEngine.CustomerType.REGULAR,
                true, "SAVE10", true, 500
        );

        double expected = 100.0 * (1.0 - 0.35) - 5.0 + 7.99;
        double result = engine.calculateFinalPrice(request);

        assertEquals(expected, result, 0.001);
    }
}
