package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;
import com.viktor.lab4.PricingRequest;

public class QwenTest20 {

    @Test(timeout = 1000)
    public void testFinalPriceWithLoyaltyTierAndSeasonalSale() {
        PricingRequest request = new PricingRequest(
                50.0, // basePrice
                15,  // quantity
                PricingEngine.CustomerType.VIP, // customerType
                true,  // seasonalSale
                "SAVE10", // couponCode
                false, // firstOrder
                400    // loyaltyPoints
        );

        double expected = 43.15;
        double actual = PricingEngine.calculateFinalPrice(request);

        assertEquals(expected, actual, 0.01);
    }
}
