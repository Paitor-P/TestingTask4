package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.PricingEngine;

public class QwenTest9 {

    @Test(timeout=1000)
    public void testCalculateFinalPriceInvalidInput() {
        PricingRequest request = new PricingRequest(
                -1.0, // basePrice must be >= 0
                0,    // quantity must be > 0
                PricingEngine.CustomerType.REGULAR,
                false,
                "SAVE10",
                true,
                100
        );
        try {
            PricingEngine.calculateFinalPrice(request);
            fail("Expected IllegalArgumentException for negative basePrice");
        } catch (IllegalArgumentException e) {
            assertEquals("basePrice must be >= 0", e.getMessage());
        }
    }
}
