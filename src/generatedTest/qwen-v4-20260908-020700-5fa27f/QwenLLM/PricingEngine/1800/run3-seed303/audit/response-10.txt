                99.99,
                10,
                PricingEngine.CustomerType.REGULAR,
                true,
                "SAVE10",
                true,
                500
        );

        double result = pricingEngine.calculateFinalPrice(request);

        assertEquals(expected, result, 0.001);
    }
}
