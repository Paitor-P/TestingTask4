import org.junit.Test;

import com.viktor.lab4.PricingEngine;

        PricingRequest request = new PricingRequest(
                20.0,
                12,
                PricingEngine.CustomerType.REGULAR,
                false,

                true,
                200
        );
        double expected = 20.0 * 12 * (1.0 - 0.04) - 5.0 + 7.99;
        assertEquals("Expected final price", expected, pricingEngine.calculateFinalPrice(request), 0.001);
    }
}
