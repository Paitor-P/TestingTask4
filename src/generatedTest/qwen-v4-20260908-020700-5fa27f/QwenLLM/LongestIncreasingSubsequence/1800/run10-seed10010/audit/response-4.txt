package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest4 {

    @Test(timeout = 1000)
    public void testLengthOfLIS() {
        LongestIncreasingSubsequence lis = new LongestIncreasingSubsequence();
        
        // Test case 1: Empty array
        assertEquals(0, lis.lengthOfLIS(new int[]{}));
        
        // Test case 2: Single element array
        assertEquals(1, lis.lengthOfLIS(new int[]{5}));
        
        // Test case 3: Simple sequence
        assertEquals(3, lis.lengthOfLIS(new int[]{1, 3, 5}));
        
        // Test case 4: Multiple sequences
        assertEquals(4, lis.lengthOfLIS(new int[]{1, 2, 3, 4, 5}));
        
        // Test case 5: Duplicate elements
        assertEquals(3, lis.lengthOfLIS(new int[]{1, 2, 2, 3, 4}));
        
        // Test case 6: Decreasing sequence
        assertEquals(1, lis.lengthOfLIS(new int[]{5, 4, 3, 2, 1}));
        
        // Test case 7: Random sequence
        assertEquals(5, lis.lengthOfLIS(new int[]{3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5}));
    }
}
