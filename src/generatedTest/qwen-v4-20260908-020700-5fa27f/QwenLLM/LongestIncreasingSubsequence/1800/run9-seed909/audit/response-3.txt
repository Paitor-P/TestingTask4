package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.LongestIncreasingSubsequence;

public class QwenTest3 {
    @Test(timeout = 1000)
    public void testLengthOfLISWithBoundaryCases() {
        LongestIncreasingSubsequence lis = new LongestIncreasingSubsequence();

        // Test with an empty array
        assertEquals(0, lis.lengthOfLIS(new int[]{}));

        // Test with a single element
        assertEquals(1, lis.lengthOfLIS(new int[]{5}));

        // Test with a single element sequence
        assertEquals(1, lis.lengthOfLIS(new int[]{1, 2, 3, 4, 5}));

        // Test with a single decreasing element sequence
        assertEquals(1, lis.lengthOfLIS(new int[]{5, 4, 3, 2, 1}));
    }
}
