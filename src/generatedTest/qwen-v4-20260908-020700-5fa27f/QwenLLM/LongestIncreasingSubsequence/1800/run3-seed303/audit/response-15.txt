import static org.junit.Assert.assertEquals;

import java.util.Arrays;

public class QwenTest15 {

    public static class LongestIncreasingSubsequence {
        public int lengthOfLIS(int[] numbers) {
            if (numbers == null) {
                throw new IllegalArgumentException("numbers must not be null");
            }
            if (numbers.length == 0) {
                return 0;
            }

            int[] tails = new int[numbers.length];
            int size = 0;

            for (int number : numbers) {
                int index = Arrays.binarySearch(tails, 0, size, number);
                if (index < 0) {
                    index = -(index + 1);
                }
                tails[index] = number;
                if (index == size) {
                    size++;
                }
            }

            return size;
        }
    }

    @Test(timeout = 1000)
    public void testSingleElementSubsequence() {
        int[] numbers = {5};
        int expected = 1;
        assertEquals(expected, LongestIncreasingSubsequence.lengthOfLIS(numbers));
    }
}

This test method checks the behavior of the `lengthOfLIS` method when the input array contains only one element, which should return 1, as the longest increasing subsequence is the sequence itself.
