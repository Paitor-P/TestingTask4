package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest11 {

    @Test(timeout=1000)
    public void testLengthOfLIS() {
        LongestIncreasingSubsequence lis = new LongestIncreasingSubsequence();
        
        // Test case: Empty array
        assertEquals(0, lis.lengthOfLIS(new int[]{}));
        
        // Test case: Single element
        assertEquals(1, lis.lengthOfLIS(new int[]{5}));
        
        // Test case: Simple increasing sequence
        assertEquals(3, lis.lengthOfLIS(new int[]{1, 2, 3}));
        
        // Test case: Simple decreasing sequence
        assertEquals(1, lis.lengthOfLIS(new int[]{3, 2, 1}));
        
        // Test case: Mixed sequence
        assertEquals(4, lis.lengthOfLIS(new int[]{10, 9, 2, 5, 3, 7, 101, 18}));
        
        // Test case: Sequence with duplicates
        assertEquals(4, lis.lengthOfLIS(new int[]{1, 2, 2, 3, 4, 4, 4}));
        
        // Test case: Sequence with all elements the same
        assertEquals(1, lis.lengthOfLIS(new int[]{4, 4, 4, 4, 4, 4}));
        
        // Test case: Sequence with negative numbers
        assertEquals(4, lis.lengthOfLIS(new int[]{-1, -2, -3, -4, -5}));
        
        // Test case: Sequence with negative and positive numbers
        assertEquals(4, lis.lengthOfLIS(new int[]{-1, 2, -3, 4, -5, 6}));
        
        // Test case: Large sequence
        int[] largeSequence = new int[1000];
        Arrays.fill(largeSequence, 1);
        assertEquals(1000, lis.lengthOfLIS(largeSequence));
        
        // Test case: Large sequence with a single different number
        largeSequence[0] = 0;
        assertEquals(1001, lis.lengthOfLIS(largeSequence));
        
        // Test case: Large sequence with a single different number at the end
        largeSequence[999] = 2;
        assertEquals(1000, lis.lengthOfLIS(largeSequence));
        
        // Test case: Large sequence with a single different number at the beginning
        largeSequence[0] = 2;
        assertEquals(1001, lis.lengthOfLIS(largeSequence));
        
        // Test case: Large sequence with a single different number in the middle
        largeSequence[499] = 2;
        assertEquals(1000, lis.lengthOfLIS(largeSequence));
    }
}
