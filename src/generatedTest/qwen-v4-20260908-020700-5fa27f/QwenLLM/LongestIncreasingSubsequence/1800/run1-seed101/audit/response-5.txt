package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest5 {

    @Test(timeout=1000)
    public void testLengthOfLIS() {
        LongestIncreasingSubsequence lis = new LongestIncreasingSubsequence();

        // Test case 1: Normal case
        int[] nums = {10, 9, 2, 5, 3, 7, 101, 18};
        assertEquals(4, lis.lengthOfLIS(nums));

        // Test case 2: All elements are the same
        int[] nums2 = {1, 1, 1, 1, 1};
        assertEquals(1, lis.lengthOfLIS(nums2));

        // Test case 3: Single element
        int[] nums3 = {5};
        assertEquals(1, lis.lengthOfLIS(nums3));

        // Test case 4: Empty array
        int[] nums4 = {};
        assertEquals(0, lis.lengthOfLIS(nums4));

        // Test case 5: All increasing
        int[] nums5 = {1, 2, 3, 4, 5};
        assertEquals(5, lis.lengthOfLIS(nums5));

        // Test case 6: All decreasing
        int[] nums6 = {5, 4, 3, 2, 1};
        assertEquals(1, lis.lengthOfLIS(nums6));

        // Test case 7: Repeated elements
        int[] nums7 = {1, 2, 2, 3, 3, 4, 4, 5};
        assertEquals(5, lis.lengthOfLIS(nums7));
    }
}
