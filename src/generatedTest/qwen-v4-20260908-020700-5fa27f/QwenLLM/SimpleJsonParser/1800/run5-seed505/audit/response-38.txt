package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest37 {

    @Test(timeout=1000)
    public void testParseSimpleJson() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "{\"name\":\"John\", \"age\":30, \"city\":\"New York\"}";
        Object result = parser.parse(input);

        assertTrue(result instanceof List);
        List<Object> map = (List<Object>) result;
        assertEquals(3, map.size());

        assertEquals("John", map.get(0));
        assertEquals(30.0, map.get(1), 0.001);
        assertEquals("New York", map.get(2));
    }
}
