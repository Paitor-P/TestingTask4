package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest21 {

    @Test(timeout = 1000)
    public void testInvalidInput() {
        SimpleJsonParser parser = new SimpleJsonParser();

        try {
            parser.parse("null");
            fail("Expected IllegalArgumentException for null input");
        } catch (IllegalArgumentException e) {
            assertEquals("input must not be null", e.getMessage());
        }

        try {
            parser.parse("true");
            fail("Expected IllegalArgumentException for \"true\"");
        } catch (IllegalArgumentException e) {
            assertEquals("input must not be null", e.getMessage());
        }

        try {
            parser.parse("false");
            fail("Expected IllegalArgumentException for \"false\"");
        } catch (IllegalArgumentException e) {
            assertEquals("input must not be null", e.getMessage());
        }

        try {
            parser.parse("");
            fail("Expected IllegalArgumentException for empty string");
        } catch (IllegalArgumentException e) {
            assertEquals("input must not be null", e.getMessage());
        }

        try {
            parser.parse(" ");
            fail("Expected IllegalArgumentException for whitespace");
        } catch (IllegalArgumentException e) {
            assertEquals("input must not be null", e.getMessage());
        }
    }
}
