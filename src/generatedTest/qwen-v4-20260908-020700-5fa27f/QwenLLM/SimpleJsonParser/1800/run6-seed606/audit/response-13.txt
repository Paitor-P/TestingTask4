import org.junit.Test;

        String input = "123.45";
        Double expected = 123.45;
        Object result = parser.parse(input);

        // Check if the result is an instance of Number and compare its value
        if (result instanceof Number) {
            assertEquals(expected.doubleValue(), ((Number) result).doubleValue(), 1e-10);
        } else {
            fail("Parsed result is not a Number");
        }
    }
}
