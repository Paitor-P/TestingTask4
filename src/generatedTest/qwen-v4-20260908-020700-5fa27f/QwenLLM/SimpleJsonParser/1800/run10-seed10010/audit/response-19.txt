package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest19 {

    @Test(timeout = 1000)
    public void testParseSimpleObject() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "{\"name\":\"John\", \"age\":30, \"city\":\"New York\"}";
        Object result = parser.parse(input);
        assertTrue(result instanceof java.util.Map);
        Map<String, Object> map = (Map<String, Object>) result;
        assertEquals("John", map.get("name"));
        assertEquals(30, map.get("age"));
        assertEquals("New York", map.get("city"));
    }
}
