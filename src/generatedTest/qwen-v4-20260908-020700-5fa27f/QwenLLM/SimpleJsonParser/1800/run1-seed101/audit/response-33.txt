package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest33 {

    @Test(timeout = 1000)
    public void testParseSimpleObject() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "{\"key\": \"value\"}";
        Object result = parser.parse(input);
        assertEquals(Map.class, result.getClass());
        Map<Object, Object> map = (Map<Object, Object>) result; // Fixed from String to Object
        assertEquals("value", map.get("key"));
    }
}
