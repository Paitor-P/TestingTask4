package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest47 {

    @Test(timeout=1000)
    public void testParseValidJsonArray() {
        String json = "[1, 2, 3, \"hello\", true, false, null]";
        SimpleJsonParser parser = new SimpleJsonParser();
        Object result = parser.parse(json);
        assertTrue(result instanceof List<?>); // Changed to List<?> for type safety
        List<?> list = (List<?>) result; // Changed to List<?> for type safety
        assertEquals(7, list.size());
        assertEquals(1, list.get(0));
        assertEquals(2, list.get(1));
        assertEquals(3, list.get(2));
        assertEquals("hello", list.get(3));
        assertEquals(true, list.get(4));
        assertEquals(false, list.get(5));
        assertEquals(null, list.get(6));
    }
}
