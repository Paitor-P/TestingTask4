package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class QwenTest55 {

    @Test(timeout = 1000)
    public void testParseSimpleJson() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "{\"key\":\"value\", \"array\":[1, 2, 3], \"number\":42.5}";
        Object result = parser.parse(input);

        assertNotNull(result);
        assertTrue(result instanceof Map); // Corrected from Object to Map

        Map<String, Object> expected = new HashMap<>();
        expected.put("key", "value");
        expected.put("array", List.of(1, 2, 3)); // Corrected from Arrays.asList to List.of
        expected.put("number", 42.5);

        assertEquals(expected, result);
    }
}
