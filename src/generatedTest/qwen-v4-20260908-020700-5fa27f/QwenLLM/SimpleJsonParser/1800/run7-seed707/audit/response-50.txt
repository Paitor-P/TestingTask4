package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest50 {

    @Test(timeout = 1000)
    public void testParseNumberBelowThreshold() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "123.456";
        double expected = 123.456;
        double result = ((Double) parser.parse(input)).doubleValue();

        assertEquals("The parsed number should be 123.456", expected, result, 1e-9);
    }
}
