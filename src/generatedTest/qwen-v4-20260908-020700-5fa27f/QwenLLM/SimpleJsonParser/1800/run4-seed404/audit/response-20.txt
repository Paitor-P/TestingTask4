package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest18 {

    @Test(timeout = 1000)
    public void testParseNumberCloseToThreshold() {
        String input = "[1234567890.9999999999999999999]";
        SimpleJsonParser parser = new SimpleJsonParser();
        Object result = parser.parse(input);

        if (result instanceof Double) {
            assertEquals(1234567890.9999999999999999999, (double) result, 1e-10);
        } else {
            fail("Expected a Double, but got " + result.getClass().getName());
        }
    }
}
