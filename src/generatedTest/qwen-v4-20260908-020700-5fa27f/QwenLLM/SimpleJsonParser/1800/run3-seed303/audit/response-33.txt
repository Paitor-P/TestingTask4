package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest33 {

    @Test(timeout = 1000)
    public void testSimpleJsonParser() {
        SimpleJsonParser parser = new SimpleJsonParser();
        String input = "[{\"name\":\"John\", \"age\":30}, {\"name\":\"Jane\", \"age\":25}]";
        try {
            Object result = parser.parse(input);
            assertNotNull(result);
            assertTrue(result instanceof List);
            List<Object> list = (List<Object>) result;
            assertEquals(2, list.size());

            Object first = list.get(0);
            assertTrue(first instanceof Map);
            Map<?, ?> map = (Map<?, ?>) first;
            assertEquals("John", map.get("name"));
            assertEquals(30, map.get("age"));

            Object second = list.get(1);
            assertTrue(second instanceof Map);
            Map<?, ?> map2 = (Map<?, ?>) second;
            assertEquals("Jane", map2.get("name"));
            assertEquals(25, map2.get("age"));
        } catch (IllegalArgumentException e) {
            fail("Exception thrown: " + e.getMessage());
        }
    }
}
