package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;
import com.viktor.lab4.SimpleJsonParser;

public class QwenTest11 {

    @Test(timeout=1000)
    public void testParseSimpleArray() {
        SimpleJsonParser parser = new SimpleJsonParser();
        Object result = parser.parse("[1, 2, 3]");
        assertTrue(result instanceof List);
        assertEquals(3, ((List<?>) result).size());
        assertEquals(1, (int) ((List<?>) result).get(0));
        assertEquals(2, (int) ((List<?>) result).get(1));
        assertEquals(3, (int) ((List<?>) result).get(2));
    }
}
