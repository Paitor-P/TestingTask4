package com.viktor.lab4.autogen;

import org.junit.Test;
import static org.junit.Assert.*;

public class QwenTest43 {

    @Test(timeout=1000)
    public void testParseSimpleJson() {
        SimpleJsonParser parser = new SimpleJsonParser();

        // Test a simple JSON object
        String json = "{\"name\":\"John\", \"age\":30, \"city\":\"New York\"}";
        Object result = parser.parse(json);
        assertEquals("{\"name\":\"John\", \"age\":30, \"city\":\"New York\"}", result.toString());

        // Test a simple JSON array
        String jsonArray = "[1, 2, 3, 4, 5]";
        result = parser.parse(jsonArray);
        assertEquals("[1, 2, 3, 4, 5]", result.toString());

        // Test a simple JSON string
        String jsonString = "\"Hello, World!\"";
        result = parser.parse(jsonString);
        assertEquals("\"Hello, World!\"", result.toString());

        // Test a simple JSON null
        String jsonNull = "null";
        result = parser.parse(jsonNull);
        assertNull(result);
    }
}
